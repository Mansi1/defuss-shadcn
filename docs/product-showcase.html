CSS view file

/* -- Product Showcase (marketing) ----------------------------------
   Standalone 5:3 video frame. The play button and scrubber are the
   browser's own — <video controls> means zero JS and free keyboard,
   captions and fullscreen support. */

@layer components {
  .mk-showcase {
    container: mk-showcase / inline-size;
    position: relative;
    aspect-ratio: 5 / 3;
    width: 100%;
    max-width: 80rem;
    margin-inline: auto;
    overflow: hidden;
    border-radius: var(--radius-lg);
    background: var(--muted);

    /* video + poster image share the cover-fill contract */
    & video,
    & img {
      width: 100%;
      height: 100%;
      object-fit: cover;
      display: block;
      background: var(--muted);
    }
  }
}
